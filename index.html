// Function to get URL parameters function getQueryParam(param) { let urlParams = new URLSearchParams(window.location.search); return urlParams.get(param) || ""; } // Get details from URL let name = getQueryParam("name"); let partner = getQueryParam("partner"); let customMessage = getQueryParam("msg") || "You are my everything! ❤️"; let videoId = getQueryParam("video") || "dQw4w9WgXcQ"; // Default video ID // Handle name formatting let displayName = ""; if (name && partner) { displayName = `${name} & ${partner}`; } else if (name) { displayName = name; } else if (partner) { displayName = partner; } else { displayName = "My Love"; } // Display the custom love message document.getElementById("message").innerHTML = `💖 ${displayName}, ${customMessage} 💖`; // Set the YouTube video document.getElementById("loveVideo").src = `https://www.youtube.com/embed/${videoId}?autoplay=1`;
